#now let's create the function called country_trend.
country_trend <- function (val) {
    en_dset |> 
        filter(country == val) |> 
        autoplot(energy_consumption, aes(color = country)) +
        labs(
            title = sprintf("Trend Analysis of Net Energy Consumed in %s", val),
            subtitle = "Annual data from 1980 to 2024",
            x = "Year",
            caption = "This is an open source project. by Hassan S. Konneh 2026.",
            y = "Energy Consumed"
            # tag = "For Training Purposes"
            
        ) +
            theme_classic() +
            theme(
                plot.title = element_text(hjust = 0.5, face = "bold"),
                plot.subtitle = element_text(hjust = 0.5, face = "italic")
            )
}
# note, use sprintf(), and %s for text, %d for integers, %f for decimals, remember C++? 😎

Let’s call the function “country_trend”

country_trend("Sierra Leone") # Use any of the three countries

Let’s add some smoothing curve to understand our trend

country_smooth <- function (val) {
    country_trend(val) +
        geom_smooth( level = 0.95, span = 0.25) # level is the confidene band @ 0.95 as degault. Span control the curve smoothness.
}
